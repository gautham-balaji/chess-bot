# Runtime dependencies for the Flask application and engine.
#
# Versions are pinned to the environment that was empirically verified in the
# Phase 0 baseline (see baseline/environment.json). They are NOT speculative
# upgrades: this is the set under which the shipped models load and the
# application serves requests.
#
# Notebook / training-only dependencies live in requirements-notebook.txt.
#
# Supported Python: 3.12 (verified on 3.12.10). See docs/REPRODUCIBILITY.md.

# --- Deep learning -------------------------------------------------------------
# tensorflow was previously pinned to 2.15.0. That pin was wrong: TF 2.15 bundles
# Keras 2.x, whereas models/cnn_model.keras is a Keras 3 archive
# (metadata.json reports keras_version 3.11.1, and the archive layout is the
# Keras 3 format: config.json + metadata.json + model.weights.h5). A Keras 2
# runtime cannot read it, so a clean install from the old file produced a
# non-working application.
tensorflow==2.21.0
keras==3.13.2

# --- Classical ML --------------------------------------------------------------
# Required to unpickle models/weight_model.pkl (the Ridge weight model).
# NOTE: the pickles were serialised with scikit-learn 1.7.1. Loading them under
# 1.8.0 emits InconsistentVersionWarning. The Ridge coefficients were verified
# byte-identical to the Phase 0 baseline under 1.8.0; see docs/REPRODUCIBILITY.md.
scikit-learn==1.8.0
scipy==1.17.1
numpy==2.4.3

# --- Chess ---------------------------------------------------------------------
# The distribution is `chess` (imported as `chess`). The older `python-chess`
# name is a legacy alias whose releases stop at 1.999, so `python-chess==1.11.2`
# - which earlier revisions of this file implied - is NOT installable.
chess==1.11.2

# --- Web -----------------------------------------------------------------------
flask==3.1.3
flask-cors==6.0.2

# --- Deployment ----------------------------------------------------------------
# Used by Procfile (`web: gunicorn app:app`). Linux/macOS only - gunicorn does
# not run on Windows, and it is intentionally NOT installed in the local
# development venv, so this pin is unverified locally.
gunicorn==23.0.0
